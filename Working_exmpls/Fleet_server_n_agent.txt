import mysql.connector
import random
import time
from datetime import datetime

# -----------------------------
# MySQL Connection
# -----------------------------
conn = mysql.connector.connect(
    host="127.0.0.1",
    port=3306,
    user="root",          # change if needed
    password="YOUR_PASSWORD"
)

cursor = conn.cursor()

# -----------------------------
# Create Database
# -----------------------------
cursor.execute("CREATE DATABASE IF NOT EXISTS demo_observability")
cursor.execute("USE demo_observability")

# -----------------------------
# Create Table
# -----------------------------
cursor.execute("""
CREATE TABLE IF NOT EXISTS orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_name VARCHAR(100),
    product_name VARCHAR(100),
    quantity INT,
    price DECIMAL(10,2),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
)
""")

conn.commit()

print("Database and table ready.")

# -----------------------------
# Sample Data
# -----------------------------
customers = [
    "Ajay", "John", "Alice", "Bob",
    "Maria", "David", "Emma"
]

products = [
    "Laptop", "Phone", "Keyboard",
    "Mouse", "Monitor", "Tablet"
]

# -----------------------------
# Infinite Loop
# -----------------------------
while True:
    try:
        # ---------------------------------
        # INSERT
        # ---------------------------------
        customer = random.choice(customers)
        product = random.choice(products)
        quantity = random.randint(1, 5)
        price = round(random.uniform(50, 2000), 2)

        insert_query = """
        INSERT INTO orders
        (customer_name, product_name, quantity, price)
        VALUES (%s, %s, %s, %s)
        """

        cursor.execute(insert_query, (
            customer,
            product,
            quantity,
            price
        ))

        conn.commit()

        print(f"[{datetime.now()}] INSERTED order")

        # ---------------------------------
        # SELECT
        # ---------------------------------
        cursor.execute("""
        SELECT * FROM orders
        ORDER BY id DESC
        LIMIT 5
        """)

        rows = cursor.fetchall()

        print("LATEST ORDERS:")
        for row in rows:
            print(row)

        # ---------------------------------
        # UPDATE
        # ---------------------------------
        cursor.execute("""
        UPDATE orders
        SET quantity = quantity + 1
        WHERE id = (
            SELECT id FROM (
                SELECT id
                FROM orders
                ORDER BY RAND()
                LIMIT 1
            ) x
        )
        """)

        conn.commit()

        print("UPDATED one random order")

        # ---------------------------------
        # DELETE
        # ---------------------------------
        cursor.execute("""
        DELETE FROM orders
        ORDER BY id ASC
        LIMIT 1
        """)

        conn.commit()

        print("DELETED oldest order")

        print("-" * 60)

        # ---------------------------------
        # Sleep
        # ---------------------------------
        time.sleep(5)

    except Exception as e:
        print("ERROR:", e)
        time.sleep(5)
