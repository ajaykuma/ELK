--Using curl and reading from JSON

--create a json 'products.json'
{ "index" : { "_index" : "products", "_id" : "1" } }
{ "title" : "Nike Running Shoes" }
{ "index" : { "_index" : "products", "_id" : "2" } }
{ "title" : "Adidas Basketball Shoes" }

--use curl command
curl -s -H "Content-Type: application/json" -XPOST 'http://ch1:9200/_bulk' --data-binary @products.json
