--Creating an index that tracks errors
PUT /system_logs
{
  "mappings": {
    "properties": {
      "event_name": {
        "type": "text"
      },
      "severity": {
        "type": "keyword"
      },
      "description": {
        "type": "text"
      },
      "error_code": {
        "type": "keyword"
      },
      "service": {
        "type": "keyword"
      },
      "host": {
        "type": "keyword"
      },
      "environment": {
        "type": "keyword"
      },
      "occurrences": {
        "type": "integer"
      },
      "timestamp": {
        "type": "date"
      },
      "categories": {
        "type": "nested",
        "properties": {
          "name": {
            "type": "keyword"
          }
        }
      },
      "tags": {
        "type": "keyword"
      }
    }
  },
  "settings": {
    "number_of_shards": 2,
    "number_of_replicas": 1
  }
}
